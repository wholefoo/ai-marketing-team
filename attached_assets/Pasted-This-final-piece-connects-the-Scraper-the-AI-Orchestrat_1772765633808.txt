### The Main Application (`main.py`)

This script uses an asynchronous approach to ensure all five marketing agents run simultaneously, significantly reducing the time it takes to generate the final report.

```python
import asyncio
from scraper import scrape_marketing_data
from pdf_generator import generate_audit_report
# Assuming you have a wrapper for your LLM (OpenAI or Anthropic)
from ai_agents import launch_marketing_agent 

async def run_full_marketing_audit(target_url):
    print(f"--- Starting Discovery for {target_url} ---")
    
    # [cite_start]Phase 1: Discovery & Pre-analysis [cite: 228]
    raw_data = scrape_marketing_data(target_url)
    if "error" in raw_data:
        return f"Error: {raw_data['error']}"

    # [cite_start]Phase 2: Parallel Sub-Agent Execution [cite: 242]
    [cite_start]print("Launching 5 parallel marketing sub-agents...") [cite: 37]
    
    # [cite_start]We define the 5 specific roles mentioned in the transcript [cite: 247-251]
    tasks = [
        launch_marketing_agent("Content & Messaging", raw_data),
        launch_marketing_agent("Conversion Optimization", raw_data),
        launch_marketing_agent("SEO & Discoverability", raw_data),
        launch_marketing_agent("Competitive Intelligence", raw_data),
        launch_marketing_agent("Technical Strategy", raw_data)
    ]
    
    # [cite_start]Run all agents at once [cite: 243, 244]
    agent_results = await asyncio.gather(*tasks)
    
    # [cite_start]Phase 3: Synthesis & Scoring [cite: 328]
    # One final call to synthesize the 5 reports into a single cohesive audit
    final_synthesis = await launch_marketing_agent("Lead Strategist", agent_results)
    
    # [cite_start]Phase 4: Generate Client-Ready PDF [cite: 7, 348]
    report_path = generate_audit_report(final_synthesis)
    
    [cite_start]print(f"Success! Audit complete. Report saved as: {report_path}") [cite: 369]
    return report_path
```
